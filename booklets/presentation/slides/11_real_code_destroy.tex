num_remove = max(1, int(len(
    route.client_indices) * 
    random.uniform(0.1, 0.2)))

    range(len(route.client_indices)), 
    num_remove)

for idx in sorted(indices_to_remove,

    client_idx = \
      route.client_indices.pop(idx)

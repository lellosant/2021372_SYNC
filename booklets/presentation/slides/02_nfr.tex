\begin{frame}{Non-Functional Requirements}
\begin{itemize}
    \item \textbf{Usability}\\
    The application should be easy to use and understandable without specific technical knowledge.
    
    \vspace{0.3cm}
    \item \textbf{Reusability}\\
    The application should be reusable with future ERP files having the same logical structure.
    
    \vspace{0.3cm}
    \item \textbf{Portability}\\
    The application should be deployable on another computer using Docker and Docker Compose.
    
    \vspace{0.3cm}
    \item \textbf{Accuracy \& Robustness}\\
    Provide clear error messages for invalid files and use real geographic coordinates for customer locations.
\end{itemize}
\end{frame}

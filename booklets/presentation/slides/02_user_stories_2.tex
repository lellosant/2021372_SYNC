\begin{frame}{Main User Stories (2/2)}
\begin{itemize}
    \item \textbf{Start Date \& Working Days}\\
    \textit{"As a Sales Manager, I want to choose the starting date of the visit campaign, so that I can plan visits for a specific commercial period."}
    
    \vspace{0.3cm}
    \item \textbf{Suggested Visit Sequence}\\
    \textit{"As a Sales Manager, I want the system to create a suggested sequence of customer visits, so that I can organize the working day."}
    
    \vspace{0.3cm}
    \item \textbf{Recoverable Revenue}\\
    \textit{"As a Sales Manager, I want to see the estimated recoverable revenue of the proposed plan, so that I can understand the economic value."}
    
    \vspace{0.3cm}
    \item \textbf{What-if Analysis}\\
    \textit{"As a Sales Manager, I want to change the number of available days and run the analysis again, so that I can compare different scenarios."}
\end{itemize}
\end{frame}

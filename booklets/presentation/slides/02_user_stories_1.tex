\begin{frame}{Main User Stories (1/2)}
\begin{itemize}
    \item \textbf{Upload \& Reuse}\\
    \textit{"As a Business Analyst, I want to upload an Excel file, so that I can work with the latest data extracted from the ERP."}
    
    \vspace{0.4cm}
    \item \textbf{Company Selection}\\
    \textit{"As a Sales Manager, I want to select a company, so that I can analyze only the customers and revenue related to that company."}
    
    \vspace{0.4cm}
    \item \textbf{Customer Map}\\
    \textit{"As a Sales Manager, I want to see the customers of the selected company on a map, so that I can understand their geographic distribution."}
\end{itemize}
\end{frame}

\begin{frame}{Optimization Problem}
\small
Finding the exact global optimum is \textbf{NP-hard} (Team Orienteering with Time Windows).\\
An exhaustive search is computationally intractable, requiring \textbf{heuristic optimization}.

\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{INPUTS}
\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}
    \item ERP Customers \& Revenues
    \item Starting Depot location
    \item Available business days ($N$)
    \item Target visit duration (2h--4h)
    \item OSRM travel duration matrix
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{CONSTRAINTS}
\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}
    \item \textbf{Calendar}: Mon--Fri (skip holidays)
    \item \textbf{Work budget}: max 8h/day (09--18)
    \item \textbf{Lunch slot}: flexible 1h window
    \item \textbf{Uniqueness}: at most 1 visit/client
    \item \textbf{Transfers}: multi-day trips for remote clients
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.95cm}
\small
\textbf{Objective}: \textbf{Maximize total recovered revenue} while guaranteeing feasibility.
\end{frame}

\begin{frame}[fragile]{Docker Deployment}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\vspace{0.2cm}
\textbf{Docker}\\
Packages the application and its dependencies.

\vspace{0.25cm}
\textbf{Docker Compose}\\
Orchestrates multi-container lifecycle.

\vspace{0.25cm}
\textbf{Frontend}: Port 8080 (Web UI)\\
\textbf{Backend}: Port 8000 (REST API)

\vspace{0.25cm}
\begin{block}{Separation of Concerns}
\scriptsize
Separate host ports \textbf{isolate} the end-user Web UI from direct Swagger/REST API debugging.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\vspace{0.2cm}
\begin{block}{docker-compose.yml}
\scriptsize
\begin{verbatim}
backend:
  build: ./backend
  ports: ["8000:8000"]

frontend:
  build: ./frontend
  ports: ["8080:80"]
  depends_on: [backend]
\end{verbatim}
\end{block}
\end{column}
\end{columns}
\end{frame}
